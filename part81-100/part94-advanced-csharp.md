## ขั้นตอนที่ 931: Source Generators

### แนวคิด Source Generators คืออะไร?

Source Generators เป็นฟีเจอร์ที่ถูกเพิ่มใน C# 9.0 ซึ่งช่วยให้นักพัฒนาสามารถสร้างโค้ด C# แบบอัตโนมัติในระหว่างการ compile ได้ โดยใช้ Roslyn API ซึ่งเป็น compiler platform ของ Microsoft

**ประโยชน์ของ Source Generators:**
- สร้าง boilerplate code อัตโนมัติ (เช่น ToString(), GetHashCode(), Equals())
- ลดการใช้ Reflection ที่ส่งผลต่อประสิทธิภาพในขณะ runtime
- ตรวจสอบ syntax ได้ตั้งแต่ compile time
- รองรับ IntelliSense และ debugging เต็มรูปแบบ

### ตัวอย่างที่ 1: สร้าง Source Generator อย่างง่าย

```csharp
// AutoPropertyGenerator.cs - Source Generator สำหรับสร้าง property อัตโนมัติ
using Microsoft.CodeAnalysis;
using Microsoft.CodeAnalysis.CSharp.Syntax;
using System.Collections.Generic;
using System.Linq;
using System.Text;

[Generator]
public class AutoToStringGenerator : ISourceGenerator
{
    public void Initialize(GeneratorInitializationContext context)
    {
        // ลงทะเบียน syntax receiver เพื่อดักจับ class ที่มี attribute ที่ต้องการ
        context.RegisterForSyntaxNotifications(() => new AutoToStringSyntaxReceiver());
    }

    public void Execute(GeneratorExecutionContext context)
    {
        // ตรวจสอบว่ามี syntax receiver หรือไม่
        if (context.SyntaxReceiver is not AutoToStringSyntaxReceiver receiver)
            return;

        // เพิ่ม attribute definition
        context.AddSource("AutoToStringAttribute.g.cs", @"
using System;

[AttributeUsage(AttributeTargets.Class)]
public class AutoToStringAttribute : Attribute { }
");

        // สร้างโค้ดสำหรับแต่ละ class ที่มี [AutoToString] attribute
        foreach (var classDeclaration in receiver.CandidateClasses)
        {
            var semanticModel = context.Compilation.GetSemanticModel(classDeclaration.SyntaxTree);
            var classSymbol = semanticModel.GetDeclaredSymbol(classDeclaration) as INamedTypeSymbol;

            if (classSymbol == null) continue;

            // ตรวจสอบว่า class มี [AutoToString] attribute จริงๆ
            var hasAttribute = classSymbol.GetAttributes()
                .Any(attr => attr.AttributeClass?.Name == "AutoToStringAttribute");

            if (!hasAttribute) continue;

            // สร้าง source code
            var source = GenerateToStringMethod(classSymbol);
            context.AddSource($"{classSymbol.Name}_AutoToString.g.cs", source);
        }
    }

    private string GenerateToStringMethod(INamedTypeSymbol classSymbol)
    {
        var namespaceName = classSymbol.ContainingNamespace.IsGlobalNamespace
            ? string.Empty
            : classSymbol.ContainingNamespace.ToDisplayString();

        // เก็บรายการ properties ทั้งหมด
        var properties = classSymbol.GetMembers()
            .OfType<IPropertySymbol>()
            .Where(p => p.DeclaredAccessibility == Accessibility.Public)
            .ToList();

        var sb = new StringBuilder();
        sb.AppendLine("// <auto-generated/>");
        sb.AppendLine("using System.Text;");
        sb.AppendLine();

        if (!string.IsNullOrEmpty(namespaceName))
        {
            sb.AppendLine($"namespace {namespaceName}");
            sb.AppendLine("{");
        }

        sb.AppendLine($"    public partial class {classSymbol.Name}");
        sb.AppendLine("    {");
        sb.AppendLine("        public override string ToString()");
        sb.AppendLine("        {");
        sb.AppendLine("            var sb = new StringBuilder();");
        sb.AppendLine($"            sb.Append(\"{classSymbol.Name} {{ \");");

        for (int i = 0; i < properties.Count; i++)
        {
            var prop = properties[i];
            var separator = i < properties.Count - 1 ? ", " : "";
            sb.AppendLine($"            sb.Append(\"{prop.Name} = \" + {prop.Name} + \"{separator}\");");
        }

        sb.AppendLine("            sb.Append(\" }\");");
        sb.AppendLine("            return sb.ToString();");
        sb.AppendLine("        }");
        sb.AppendLine("    }");

        if (!string.IsNullOrEmpty(namespaceName))
        {
            sb.AppendLine("}");
        }

        return sb.ToString();
    }
}

// Syntax Receiver สำหรับดักจับ class declarations
public class AutoToStringSyntaxReceiver : ISyntaxReceiver
{
    public List<ClassDeclarationSyntax> CandidateClasses { get; } = new();

    public void OnVisitSyntaxNode(SyntaxNode syntaxNode)
    {
        // เพิ่มเฉพาะ class ที่มี attribute
        if (syntaxNode is ClassDeclarationSyntax classDeclaration
            && classDeclaration.AttributeLists.Count > 0)
        {
            CandidateClasses.Add(classDeclaration);
        }
    }
}
```

### ตัวอย่างที่ 3: Incremental Source Generator (แนะนำสำหรับ .NET 6+)

```csharp
// IncrementalSourceGenerator.cs - เวอร์ชันที่มีประสิทธิภาพสูงกว่า
using Microsoft.CodeAnalysis;
using Microsoft.CodeAnalysis.CSharp.Syntax;
using System.Collections.Immutable;
using System.Text;

[Generator]
public class SerializableGenerator : IIncrementalGenerator
{
    public void Initialize(IncrementalGeneratorInitializationContext context)
    {
        // สร้าง pipeline สำหรับหา class ที่มี [Serializable] attribute
        var classDeclarations = context.SyntaxProvider
            .CreateSyntaxProvider(
                predicate: static (node, _) => IsClassWithAttribute(node),
                transform: static (ctx, _) => GetClassInfo(ctx))
            .Where(static info => info is not null);

        // รวม compilation กับ class declarations
        var compilationAndClasses = context.CompilationProvider
            .Combine(classDeclarations.Collect());

        // สร้าง source
        context.RegisterSourceOutput(
            compilationAndClasses,
            static (spc, source) => Execute(source.Left, source.Right, spc));
    }

    private static bool IsClassWithAttribute(SyntaxNode node)
    {
        return node is ClassDeclarationSyntax c && c.AttributeLists.Count > 0;
    }

    private static ClassDeclarationSyntax? GetClassInfo(GeneratorSyntaxContext context)
    {
        var classDeclaration = (ClassDeclarationSyntax)context.Node;
        
        foreach (var attributeList in classDeclaration.AttributeLists)
        {
            foreach (var attribute in attributeList.Attributes)
            {
                var symbol = context.SemanticModel.GetSymbolInfo(attribute).Symbol;
                if (symbol?.ContainingType?.Name == "JsonSerializableAttribute")
                    return classDeclaration;
            }
        }
        
        return null;
    }

    private static void Execute(
        Compilation compilation,
        ImmutableArray<ClassDeclarationSyntax?> classes,
        SourceProductionContext context)
    {
        foreach (var classDeclaration in classes)
        {
            if (classDeclaration is null) continue;
            
            var semanticModel = compilation.GetSemanticModel(classDeclaration.SyntaxTree);
            var classSymbol = semanticModel.GetDeclaredSymbol(classDeclaration) as INamedTypeSymbol;
            
            if (classSymbol is null) continue;

            var source = GenerateSerializer(classSymbol);
            context.AddSource($"{classSymbol.Name}_Serializer.g.cs", source);
        }
    }

    private static string GenerateSerializer(INamedTypeSymbol classSymbol)
    {
        var sb = new StringBuilder();
        sb.AppendLine("// <auto-generated/>");
        sb.AppendLine("using System.Text.Json;");
        sb.AppendLine();
        sb.AppendLine($"public static class {classSymbol.Name}Serializer");
        sb.AppendLine("{");
        sb.AppendLine($"    public static string Serialize({classSymbol.Name} obj)");
        sb.AppendLine("        => JsonSerializer.Serialize(obj);");
        sb.AppendLine();
        sb.AppendLine($"    public static {classSymbol.Name}? Deserialize(string json)");
        sb.AppendLine($"        => JsonSerializer.Deserialize<{classSymbol.Name}>(json);");
        sb.AppendLine("}");
        
        return sb.ToString();
    }
}
```

## ขั้นตอนที่ 932: Expression Trees

### แนวคิด Expression Trees คืออะไร?

Expression Trees คือโครงสร้างข้อมูลที่แทน code ในรูปแบบ tree ของ objects โดยแต่ละ node ใน tree จะแทนการดำเนินการทางโปรแกรม เช่น การเรียกเมธอด การดำเนินการทางคณิตศาสตร์ หรือการเปรียบเทียบ

**การใช้งานทั่วไป:**
- สร้าง dynamic LINQ queries (เช่น ใน Entity Framework)
- สร้าง lambda expressions แบบ dynamic
- ใช้ใน ORM frameworks สำหรับ query translation

### ตัวอย่างที่ 1: สร้าง Expression Trees แบบ Manual

```csharp
using System;
using System.Linq.Expressions;

class ExpressionTreeExamples
{
    static void Main()
    {
        // ตัวอย่างที่ 1: สร้าง Expression Tree แบบง่าย
        // แทน: x => x + 5
        var parameter = Expression.Parameter(typeof(int), "x");
        var constant = Expression.Constant(5, typeof(int));
        var addExpression = Expression.Add(parameter, constant);
        var lambda = Expression.Lambda<Func<int, int>>(addExpression, parameter);

        // Compile และเรียกใช้
        var compiled = lambda.Compile();
        Console.WriteLine($"lambda(10) = {compiled(10)}"); // ผลลัพธ์: 15
        Console.WriteLine($"Expression: {lambda}");         // ผลลัพธ์: x => (x + 5)

        Console.WriteLine();

        // ตัวอย่างที่ 2: Expression Tree สำหรับเปรียบเทียบ
        // แทน: (x, y) => x > y
        var xParam = Expression.Parameter(typeof(int), "x");
        var yParam = Expression.Parameter(typeof(int), "y");
        var greaterThan = Expression.GreaterThan(xParam, yParam);
        var compareLambda = Expression.Lambda<Func<int, int, bool>>(greaterThan, xParam, yParam);

        var compareFunc = compareLambda.Compile();
        Console.WriteLine($"10 > 5 = {compareFunc(10, 5)}"); // true
        Console.WriteLine($"3 > 7 = {compareFunc(3, 7)}");   // false

        Console.WriteLine();

        // ตัวอย่างที่ 3: Expression Tree สำหรับ object property
        DemonstrateMemberAccess();
    }

    static void DemonstrateMemberAccess()
    {
        // สร้าง Expression ที่ access property ของ object
        // แทน: (Person p) => p.Name
        var personParam = Expression.Parameter(typeof(Person), "p");
        var nameProperty = Expression.Property(personParam, nameof(Person.Name));
        var getNameLambda = Expression.Lambda<Func<Person, string>>(nameProperty, personParam);

        var getName = getNameLambda.Compile();

        var person = new Person { Name = "สมหญิง", Age = 25 };
        Console.WriteLine($"Name: {getName(person)}"); // สมหญิง
    }
}

public class Person
{
    public string Name { get; set; } = "";
    public int Age { get; set; }
}
```

### ตัวอย่างที่ 2: Dynamic Query Builder ด้วย Expression Trees

```csharp
using System;
using System.Collections.Generic;
using System.Linq;
using System.Linq.Expressions;

// Dynamic Query Builder สำหรับสร้าง WHERE clause แบบ dynamic
public class DynamicQueryBuilder<T>
{
    // สร้าง predicate สำหรับกรองข้อมูลตาม property name และ value
    public static Expression<Func<T, bool>> BuildEqualsPredicate(
        string propertyName, 
        object value)
    {
        var parameter = Expression.Parameter(typeof(T), "x");
        var property = Expression.Property(parameter, propertyName);
        var constant = Expression.Constant(value, property.Type);
        var equals = Expression.Equal(property, constant);
        
        return Expression.Lambda<Func<T, bool>>(equals, parameter);
    }

    // สร้าง predicate สำหรับ string contains
    public static Expression<Func<T, bool>> BuildContainsPredicate(
        string propertyName, 
        string searchValue)
    {
        var parameter = Expression.Parameter(typeof(T), "x");
        var property = Expression.Property(parameter, propertyName);
        
        // เรียก string.Contains method
        var containsMethod = typeof(string).GetMethod("Contains", new[] { typeof(string) })!;
        var searchConstant = Expression.Constant(searchValue);
        var containsCall = Expression.Call(property, containsMethod, searchConstant);
        
        return Expression.Lambda<Func<T, bool>>(containsCall, parameter);
    }

    // รวม predicates ด้วย AND
    public static Expression<Func<T, bool>> AndAlso(
        Expression<Func<T, bool>> left,
        Expression<Func<T, bool>> right)
    {
        var parameter = Expression.Parameter(typeof(T), "x");
        
        // แทน parameter ใน left expression ด้วย parameter ใหม่
        var leftBody = ReplaceParameter(left.Body, left.Parameters[0], parameter);
        var rightBody = ReplaceParameter(right.Body, right.Parameters[0], parameter);
        
        var combined = Expression.AndAlso(leftBody, rightBody);
        return Expression.Lambda<Func<T, bool>>(combined, parameter);
    }

    // Helper: แทนที่ parameter ใน expression
    private static Expression ReplaceParameter(
        Expression expression, 
        ParameterExpression oldParam, 
        ParameterExpression newParam)
    {
        return new ParameterReplacer(oldParam, newParam).Visit(expression);
    }
}

// Helper class สำหรับแทนที่ parameter
public class ParameterReplacer : ExpressionVisitor
{
    private readonly ParameterExpression _oldParam;
    private readonly ParameterExpression _newParam;

    public ParameterReplacer(ParameterExpression oldParam, ParameterExpression newParam)
    {
        _oldParam = oldParam;
        _newParam = newParam;
    }

    protected override Expression VisitParameter(ParameterExpression node)
    {
        return node == _oldParam ? _newParam : base.VisitParameter(node);
    }
}

// ตัวอย่างการใช้งาน
class DynamicQueryDemo
{
    static void Main()
    {
        var employees = new List<Employee>
        {
            new() { Name = "สมชาย", Department = "IT", Salary = 50000 },
            new() { Name = "สมหญิง", Department = "HR", Salary = 45000 },
            new() { Name = "วิชัย", Department = "IT", Salary = 60000 },
            new() { Name = "มาลี", Department = "Finance", Salary = 55000 },
            new() { Name = "ประยุทธ์", Department = "IT", Salary = 70000 }
        };

        // สร้าง predicate แบบ dynamic
        var itFilter = DynamicQueryBuilder<Employee>
            .BuildEqualsPredicate("Department", "IT");
        
        var nameFilter = DynamicQueryBuilder<Employee>
            .BuildContainsPredicate("Name", "ส");
        
        // รวม predicates
        var combinedFilter = DynamicQueryBuilder<Employee>
            .AndAlso(itFilter, nameFilter);

        // Query
        var result = employees.AsQueryable().Where(combinedFilter).ToList();

        Console.WriteLine("พนักงานแผนก IT ที่ชื่อขึ้นต้นด้วย 'ส':");
        foreach (var emp in result)
        {
            Console.WriteLine($"  - {emp.Name} ({emp.Department}): {emp.Salary:N0} บาท");
        }
        // ผลลัพธ์: สมชาย (IT): 50,000 บาท
    }
}

public class Employee
{
    public string Name { get; set; } = "";
    public string Department { get; set; } = "";
    public decimal Salary { get; set; }
}
```

### ตัวอย่างที่ 3: Expression Tree Visitor

```csharp
using System;
using System.Linq.Expressions;

// ExpressionVisitor สำหรับแปลง Expression เป็น SQL-like string
public class SqlLikeTranslator : ExpressionVisitor
{
    private System.Text.StringBuilder _sb = new();

    public string Translate<T>(Expression<Func<T, bool>> expression)
    {
        _sb.Clear();
        Visit(expression.Body);
        return _sb.ToString();
    }

    protected override Expression VisitBinary(BinaryExpression node)
    {
        _sb.Append("(");
        Visit(node.Left);
        
        _sb.Append(node.NodeType switch
        {
            ExpressionType.Equal => " = ",
            ExpressionType.NotEqual => " <> ",
            ExpressionType.GreaterThan => " > ",
            ExpressionType.GreaterThanOrEqual => " >= ",
            ExpressionType.LessThan => " < ",
            ExpressionType.LessThanOrEqual => " <= ",
            ExpressionType.AndAlso => " AND ",
            ExpressionType.OrElse => " OR ",
            _ => throw new NotSupportedException($"ไม่รองรับ operator: {node.NodeType}")
        });
        
        Visit(node.Right);
        _sb.Append(")");
        
        return node;
    }

    protected override Expression VisitMember(MemberExpression node)
    {
        if (node.Expression is ParameterExpression)
        {
            _sb.Append(node.Member.Name);
        }
        return node;
    }

    protected override Expression VisitConstant(ConstantExpression node)
    {
        if (node.Value is string strValue)
            _sb.Append($"'{strValue}'");
        else
            _sb.Append(node.Value);
        
        return node;
    }
}

// การทดสอบ
class SqlTranslatorDemo
{
    static void Main()
    {
        var translator = new SqlLikeTranslator();

        Expression<Func<Employee, bool>> expr1 = 
            e => e.Department == "IT" && e.Salary > 50000;
        
        Console.WriteLine($"Expression: {expr1}");
        Console.WriteLine($"SQL-like: {translator.Translate(expr1)}");
        // ผลลัพธ์: ((Department = 'IT') AND (Salary > 50000))

        Expression<Func<Employee, bool>> expr2 = 
            e => e.Name == "สมชาย" || e.Department == "HR";
        
        Console.WriteLine($"\nExpression: {expr2}");
        Console.WriteLine($"SQL-like: {translator.Translate(expr2)}");
        // ผลลัพธ์: ((Name = 'สมชาย') OR (Department = 'HR'))
    }
}
```

## ขั้นตอนที่ 933: Dynamic Programming

### แนวคิด Dynamic Language Runtime (DLR)

DLR (Dynamic Language Runtime) เป็น runtime environment ที่ช่วยให้ C# รองรับ dynamic typing ผ่าน keyword `dynamic` โดยการตัดสินใจเกี่ยวกับ type จะเกิดขึ้นที่ runtime แทน compile time

### ตัวอย่างที่ 1: ExpandoObject


### ตัวอย่างที่ 2: DynamicObject

```csharp
using System;
using System.Dynamic;
using System.Collections.Generic;

// Custom DynamicObject ที่เก็บ state ใน Dictionary
public class DynamicRecord : DynamicObject
{
    private readonly Dictionary<string, object?> _data = new();
    private readonly List<string> _accessLog = new();

    // Override สำหรับการ get property
    public override bool TryGetMember(GetMemberBinder binder, out object? result)
    {
        _accessLog.Add($"GET: {binder.Name}");
        
        if (_data.TryGetValue(binder.Name, out result))
            return true;
        
        // ถ้าไม่มี property ให้ return null แทนที่จะ throw exception
        result = null;
        return true;
    }

    // Override สำหรับการ set property
    public override bool TrySetMember(SetMemberBinder binder, object? value)
    {
        _accessLog.Add($"SET: {binder.Name} = {value}");
        _data[binder.Name] = value;
        return true;
    }

    // Override สำหรับการเรียก method
    public override bool TryInvokeMember(
        InvokeMemberBinder binder, 
        object?[]? args, 
        out object? result)
    {
        _accessLog.Add($"INVOKE: {binder.Name}({string.Join(", ", args ?? Array.Empty<object?>())})");
        
        // ตรวจสอบว่ามี method ที่เก็บไว้หรือไม่
        if (_data.TryGetValue(binder.Name, out var method) && method is Delegate del)
        {
            result = del.DynamicInvoke(args);
            return true;
        }

        result = null;
        return false;
    }

    // แสดง access log
    public void PrintAccessLog()
    {
```

```csharp
using System;
using System.IO;
using System.Reflection;
using System.Runtime.Loader;

// Custom AssemblyLoadContext สำหรับ plugin system
public class PluginLoadContext : AssemblyLoadContext
{
    private readonly AssemblyDependencyResolver _resolver;
    private readonly string _pluginPath;

    public PluginLoadContext(string pluginPath) : base(isCollectible: true)
    {
        _pluginPath = pluginPath;
        _resolver = new AssemblyDependencyResolver(pluginPath);
    }

    protected override Assembly? Load(AssemblyName assemblyName)
    {
        // ลองหา assembly จาก plugin directory ก่อน
        var assemblyPath = _resolver.ResolveAssemblyToPath(assemblyName);
        if (assemblyPath != null)
            return LoadFromAssemblyPath(assemblyPath);

        // ถ้าไม่พบ ให้ใช้ default context
        return null;
    }

    protected override IntPtr LoadUnmanagedDll(string unmanagedDllName)
    {
        var libraryPath = _resolver.ResolveUnmanagedDllToPath(unmanagedDllName);
        if (libraryPath != null)
            return LoadUnmanagedDllFromPath(libraryPath);

        return IntPtr.Zero;
    }
}

// Plugin Interface
public interface IPlugin
{
    string Name { get; }
    string Version { get; }
    void Execute(string input);
}

// Plugin Manager
public class PluginManager
{
    private readonly Dictionary<string, (PluginLoadContext Context, IPlugin Plugin)> _plugins = new();

    public IPlugin? LoadPlugin(string pluginPath)
    {
        try
        {
            var context = new PluginLoadContext(pluginPath);
            var assembly = context.LoadFromAssemblyPath(pluginPath);
            
            // หา type ที่ implement IPlugin
            var pluginType = assembly.GetTypes()
                .FirstOrDefault(t => typeof(IPlugin).IsAssignableFrom(t) && !t.IsInterface);

            if (pluginType == null)
            {
                Console.WriteLine($"ไม่พบ IPlugin implementation ใน {Path.GetFileName(pluginPath)}");
                context.Unload();
                return null;
            }

            var plugin = (IPlugin?)Activator.CreateInstance(pluginType);
            if (plugin != null)
            {
                _plugins[plugin.Name] = (context, plugin);
                Console.WriteLine($"โหลด plugin: {plugin.Name} v{plugin.Version}");
                return plugin;
            }

            context.Unload();
            return null;
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Error loading plugin: {ex.Message}");
            return null;
        }
    }

    public void UnloadPlugin(string pluginName)
    {
        if (_plugins.TryGetValue(pluginName, out var entry))
        {
            _plugins.Remove(pluginName);
            entry.Context.Unload();
            Console.WriteLine($"Unloaded plugin: {pluginName}");
        }
    }

    public IPlugin? GetPlugin(string name) =>
        _plugins.TryGetValue(name, out var entry) ? entry.Plugin : null;

    public IEnumerable<IPlugin> GetAllPlugins() =>
        _plugins.Values.Select(v => v.Plugin);
}
```

```csharp
using System;
using System.Collections.Generic;

// Covariance (out) - ใช้ใน return type เท่านั้น
// IEnumerable<T> เป็น covariant ใน T
public interface IProducer<out T>
{
    T Produce();
    IEnumerable<T> ProduceMany(int count);
}

// Contravariance (in) - ใช้ใน parameter type เท่านั้น
// Action<T> เป็น contravariant ใน T
public interface IConsumer<in T>
{
    void Consume(T item);
    void ConsumeMany(IEnumerable<T> items);
}

// Invariant (ไม่ใช่ทั้งสอง)
public interface IRepository<T>
{
    T Get(int id);
    void Add(T item);
}

// Implementation
public class AnimalProducer : IProducer<Animal>
{
    public Animal Produce() => new Animal { Name = "สัตว์ทั่วไป", Age = 1 };
    public IEnumerable<Animal> ProduceMany(int count) =>
        Enumerable.Range(1, count).Select(i => new Animal { Name = $"Animal{i}", Age = i });
}

public class DogProducer : IProducer<Dog>
{
    public Dog Produce() => new Dog { Name = "บัดดี้", Age = 2, Breed = "Labrador" };
    public IEnumerable<Dog> ProduceMany(int count) =>
        Enumerable.Range(1, count).Select(i => new Dog { Name = $"Dog{i}", Age = i, Breed = "Poodle" });
}

public class AnimalConsumer : IConsumer<Animal>
{
    public void Consume(Animal item) => Console.WriteLine($"Consuming: {item.Name}");
    public void ConsumeMany(IEnumerable<Animal> items)
    {
        foreach (var item in items)
            Consume(item);
    }
}

public class Dog : Animal
{
    public string Breed { get; set; } = "";
}

{
    static void Main()
    {
        Console.WriteLine("=== Covariance ===");
        
        // Covariance: IProducer<Dog> สามารถใช้เป็น IProducer<Animal> ได้
        // เพราะ Dog เป็น subtype ของ Animal
        IProducer<Dog> dogProducer = new DogProducer();
        IProducer<Animal> animalProducer = dogProducer; // OK เพราะ covariance (out)
        
        Animal animal = animalProducer.Produce();
        Console.WriteLine($"Produced: {animal.Name} อายุ {animal.Age}");

        Console.WriteLine("\n=== Contravariance ===");
        
        // Contravariance: IConsumer<Animal> สามารถใช้เป็น IConsumer<Dog> ได้
        // เพราะ consumer ที่รับ Animal ก็รับ Dog ได้ (Dog เป็น Animal)
        IConsumer<Animal> animalConsumer = new AnimalConsumer();
        IConsumer<Dog> dogConsumer = animalConsumer; // OK เพราะ contravariance (in)
        
        dogConsumer.Consume(new Dog { Name = "โกลเด้น", Age = 3, Breed = "Golden Retriever" });

        Console.WriteLine("\n=== Delegate Variance ===");
        
        // Func<T, TResult> เป็น covariant ใน TResult และ contravariant ใน T
        Func<Animal, string> describeAnimal = a => $"Animal: {a.Name}";
        
        // Covariant return: Func<Animal, Animal> -> Func<Animal, object>
        Func<Animal, object> asObject = describeAnimal; // OK
        
        // Contravariant parameter: Func<object, string> -> Func<Animal, string>
        Func<object, string> fromObject = obj => obj.ToString()!;
        Func<Animal, string> fromAnimal = fromObject; // OK
        
        Console.WriteLine(fromAnimal(new Dog { Name = "เพื่อน", Age = 1, Breed = "Beagle" }));
    }
```

```csharp
using System;

// State machine ด้วย pattern matching
enum OrderStatus { Pending, Processing, Shipped, Delivered, Cancelled, Refunded }

record Order(int Id, OrderStatus Status, decimal Amount, DateTime CreatedAt);

class OrderStateMachine
{
    // Transition rules
    static OrderStatus Transition(OrderStatus current, string action) => 
        (current, action) switch
        {
            (OrderStatus.Pending, "process") => OrderStatus.Processing,
            (OrderStatus.Processing, "ship") => OrderStatus.Shipped,
            (OrderStatus.Shipped, "deliver") => OrderStatus.Delivered,
            (OrderStatus.Pending or OrderStatus.Processing, "cancel") => OrderStatus.Cancelled,
            (OrderStatus.Delivered, "refund") => OrderStatus.Refunded,
            _ => throw new InvalidOperationException(
                $"ไม่สามารถ {action} order ที่อยู่ในสถานะ {current} ได้")
        };

    // คำนวณค่าธรรมเนียมตาม status และ amount
    static decimal CalculateFee(Order order) => order switch
    {
        { Status: OrderStatus.Cancelled, Amount: > 1000 } => order.Amount * 0.05m,
        { Status: OrderStatus.Cancelled } => 50m,
        { Status: OrderStatus.Refunded, Amount: > 5000 } => order.Amount * 0.03m,
        { Status: OrderStatus.Refunded } => order.Amount * 0.05m,
        { Status: OrderStatus.Delivered } => 0m,
        _ => 0m
    };

    static void Main()
    {
        Console.WriteLine("=== Order State Machine ===");
        var order = new Order(1001, OrderStatus.Pending, 2500m, DateTime.Now);
        
        Console.WriteLine($"Initial: Order #{order.Id} - {order.Status}");
        
        var actions = new[] { "process", "ship", "deliver" };
        var currentStatus = order.Status;
        
        foreach (var action in actions)
        {
            try
            {
                currentStatus = Transition(currentStatus, action);
                Console.WriteLine($"After '{action}': {currentStatus}");
            }
            catch (InvalidOperationException ex)
            {
                Console.WriteLine($"Error: {ex.Message}");
            }
        }

        // ทดสอบ cancel order
        Console.WriteLine("\n=== Test Cancel ===");
        var pendingOrder = order with { Status = OrderStatus.Pending };
        var cancelledStatus = Transition(pendingOrder.Status, "cancel");
        var cancelledOrder = pendingOrder with { Status = cancelledStatus };
        Console.WriteLine($"Status: {cancelledOrder.Status}");
        Console.WriteLine($"Fee: {CalculateFee(cancelledOrder):C}");

        // ทดสอบ invalid transition
        Console.WriteLine("\n=== Invalid Transition ===");

        {
            Transition(OrderStatus.Delivered, "process");

{
```

```csharp
                .Skip(page * pageSize)
                .Take(pageSize)
                .ToList();
            
            yield return items;
            page++;
        }
```

### ตัวอย่างที่ 3: Iterator Pipeline

```csharp
using System;
using System.Collections.Generic;
using System.Linq;

// Pipeline pattern ด้วย iterators
public static class IteratorPipeline
{
    // Stage 1: Generate data
    public static IEnumerable<int> Generate(int count) =>
        Enumerable.Range(1, count);

    // Stage 2: Filter
    public static IEnumerable<T> Filter<T>(
        this IEnumerable<T> source, 
        Func<T, bool> predicate)
    {
        foreach (var item in source)
        {
            if (predicate(item))
                yield return item;
        }
    }

    // Stage 3: Transform
    public static IEnumerable<TResult> Transform<T, TResult>(
        this IEnumerable<T> source, 
        Func<T, TResult> selector)
    {
        foreach (var item in source)
            yield return selector(item);
    }

    // Stage 4: Batch
    public static IEnumerable<T[]> Batch<T>(this IEnumerable<T> source, int size)
    {
        var batch = new List<T>(size);
        
        foreach (var item in source)
        {
            batch.Add(item);
            if (batch.Count == size)
            {
                yield return batch.ToArray();
                batch.Clear();
            }
        }
        
        if (batch.Count > 0)
            yield return batch.ToArray();
    }

    // Side effect แบบ lazy
    public static IEnumerable<T> Tap<T>(this IEnumerable<T> source, Action<T> action)
    {
        foreach (var item in source)
        {
            action(item);
            yield return item;
        }
    }
}

class PipelineDemo
{
    static void Main()
    {
        Console.WriteLine("=== Iterator Pipeline ===");
        
        var result = IteratorPipeline.Generate(20)
            .Filter(n => n % 2 == 0)          // เอาเฉพาะเลขคู่
            .Transform(n => n * n)              // ยกกำลังสอง
            .Filter(n => n > 50)               // เอาเฉพาะที่ > 50
            .Tap(n => Console.Write($"  Processing: {n}... "))
            .Batch(3);                          // จัดกลุ่มทีละ 3

        Console.WriteLine();
        int batchNum = 1;
        foreach (var batch in result)
        {
            Console.WriteLine($"Batch {batchNum++}: [{string.Join(", ", batch)}]");
        }
    }
}
```

### แนวคิด Operator Overloading

Operator Overloading ช่วยให้ operators ทั่วไป (+, -, *, /, ==, !=, <, >, etc.) ทำงานกับ custom types ได้

### ตัวอย่างที่ 1: Custom Numeric Type

```csharp
using System;
using System.Numerics;

// Vector2D ที่มี operator overloading เต็มรูปแบบ
public struct Vector2D : IEquatable<Vector2D>
{
    public double X { get; }
    public double Y { get; }

    public Vector2D(double x, double y)
    {
        X = x;
        Y = y;
    }

    // Arithmetic operators
    public static Vector2D operator +(Vector2D a, Vector2D b) =>
        new(a.X + b.X, a.Y + b.Y);

    public static Vector2D operator -(Vector2D a, Vector2D b) =>
        new(a.X - b.X, a.Y - b.Y);

    public static Vector2D operator -(Vector2D v) =>
        new(-v.X, -v.Y); // unary negation

    public static Vector2D operator *(Vector2D v, double scalar) =>
        new(v.X * scalar, v.Y * scalar);

    public static Vector2D operator *(double scalar, Vector2D v) =>
        v * scalar; // commutative

    public static Vector2D operator /(Vector2D v, double scalar) =>
        new(v.X / scalar, v.Y / scalar);

    // Dot product
    public static double operator *(Vector2D a, Vector2D b) =>
        a.X * b.X + a.Y * b.Y;

    // Comparison operators
    public static bool operator ==(Vector2D a, Vector2D b) =>
        a.Equals(b);

    public static bool operator !=(Vector2D a, Vector2D b) =>
        !a.Equals(b);

    // ขนาดของ vector (magnitude comparison)
    public static bool operator <(Vector2D a, Vector2D b) =>
        a.Magnitude < b.Magnitude;

    public static bool operator >(Vector2D a, Vector2D b) =>
        a.Magnitude > b.Magnitude;

    public static bool operator <=(Vector2D a, Vector2D b) =>
        a.Magnitude <= b.Magnitude;

    public static bool operator >=(Vector2D a, Vector2D b) =>
        a.Magnitude >= b.Magnitude;

    // Properties
    public double Magnitude => Math.Sqrt(X * X + Y * Y);
    public Vector2D Normalized => this / Magnitude;

    // Methods
    public double Dot(Vector2D other) => this * other;
    
    public double Cross(Vector2D other) => X * other.Y - Y * other.X;
    
    public double AngleTo(Vector2D other) =>
        Math.Acos((this * other) / (Magnitude * other.Magnitude));

    // IEquatable<T>
    public bool Equals(Vector2D other) =>
        Math.Abs(X - other.X) < 1e-10 && Math.Abs(Y - other.Y) < 1e-10;

    public override bool Equals(object? obj) =>
        obj is Vector2D other && Equals(other);

    public override int GetHashCode() => HashCode.Combine(X, Y);

    public override string ToString() => $"({X:F2}, {Y:F2})";

    // Deconstruct
    public void Deconstruct(out double x, out double y)
    {
        x = X;
        y = Y;
    }
}

{
    static void Main()
    {
        Console.WriteLine("=== Vector2D Operator Overloading ===");

        Console.WriteLine($"v2 = {v2}, magnitude = {v2.Magnitude:F2}");
        
        Console.WriteLine($"\nv1 + v2 = {v1 + v2}");
        Console.WriteLine($"v1 - v2 = {v1 - v2}");
        Console.WriteLine($"v1 * 2 = {v1 * 2}");
        Console.WriteLine($"3 * v2 = {3 * v2}");
        Console.WriteLine($"v1 / 2 = {v1 / 2}");
        Console.WriteLine($"-v1 = {-v1}");
        
        Console.WriteLine($"\nDot product v1·v2 = {v1 * v2:F2}");
        Console.WriteLine($"Cross product v1×v2 = {v1.Cross(v2):F2}");
        Console.WriteLine($"Angle between v1 and v2 = {v1.AngleTo(v2) * 180 / Math.PI:F2}°");
        
        Console.WriteLine($"\nv1 > v2: {v1 > v2}");
        Console.WriteLine($"v1 == v1: {v1 == v1}");
        Console.WriteLine($"v1 == v2: {v1 == v2}");
        
        // Deconstruction
        var (x, y) = v1;
        Console.WriteLine($"\nDeconstructed v1: x={x}, y={y}");
    }
```

```csharp
using System;

// Money type ที่มี custom conversions
public struct Money : IEquatable<Money>, IComparable<Money>
{
    public decimal Amount { get; }
    public string Currency { get; }

    public Money(decimal amount, string currency = "THB")
    {
        Amount = amount;
        Currency = currency.ToUpperInvariant();
    }

    // Implicit conversion จาก decimal (ปลอดภัย)
    public static implicit operator Money(decimal amount) =>
        new(amount, "THB");

    // Implicit conversion จาก int
    public static implicit operator Money(int amount) =>
        new(amount, "THB");

    // Explicit conversion เป็น decimal (อาจสูญเสีย currency info)
    public static explicit operator decimal(Money money) => money.Amount;

    // Explicit conversion เป็น double
    public static explicit operator double(Money money) => (double)money.Amount;

    // Arithmetic operators
    public static Money operator +(Money a, Money b)
    {
        if (a.Currency != b.Currency)
            throw new InvalidOperationException(
                $"ไม่สามารถบวก {a.Currency} กับ {b.Currency} โดยตรง");
        return new(a.Amount + b.Amount, a.Currency);
    }

    public static Money operator -(Money a, Money b)
    {
        if (a.Currency != b.Currency)
            throw new InvalidOperationException(
                $"ไม่สามารถลบ {a.Currency} กับ {b.Currency} โดยตรง");
        return new(a.Amount - b.Amount, a.Currency);
    }

    public static Money operator *(Money money, decimal factor) =>
        new(money.Amount * factor, money.Currency);

    public static Money operator *(decimal factor, Money money) =>
        money * factor;

    public static Money operator /(Money money, decimal divisor) =>
        new(money.Amount / divisor, money.Currency);

    // Comparison operators
    public static bool operator ==(Money a, Money b) => a.Equals(b);
    public static bool operator !=(Money a, Money b) => !a.Equals(b);
    public static bool operator <(Money a, Money b) => a.CompareTo(b) < 0;
    public static bool operator >(Money a, Money b) => a.CompareTo(b) > 0;
    public static bool operator <=(Money a, Money b) => a.CompareTo(b) <= 0;
    public static bool operator >=(Money a, Money b) => a.CompareTo(b) >= 0;

    // IEquatable<T>
    public bool Equals(Money other) =>
        Amount == other.Amount && Currency == other.Currency;

    public override bool Equals(object? obj) =>
        obj is Money other && Equals(other);

    public override int GetHashCode() => HashCode.Combine(Amount, Currency);

    // IComparable<T>
    public int CompareTo(Money other)
    {
        if (Currency != other.Currency)
            throw new InvalidOperationException(
                $"ไม่สามารถเปรียบเทียบ {Currency} กับ {other.Currency}");
        return Amount.CompareTo(other.Amount);
    }

    public override string ToString() => $"{Amount:N2} {Currency}";

    // Exchange rate conversion
    public Money ConvertTo(string targetCurrency, decimal rate) =>
        new(Amount * rate, targetCurrency);
}

// Temperature type
public struct Temperature
{
    private readonly double _celsius;

    private Temperature(double celsius) => _celsius = celsius;

    // Factory methods
    public static Temperature FromCelsius(double celsius) => new(celsius);
    public static Temperature FromFahrenheit(double fahrenheit) => new((fahrenheit - 32) * 5 / 9);
    public static Temperature FromKelvin(double kelvin) => new(kelvin - 273.15);

    // Properties
    public double Celsius => _celsius;
    public double Fahrenheit => _celsius * 9 / 5 + 32;
    public double Kelvin => _celsius + 273.15;

    // Arithmetic
    public static Temperature operator +(Temperature t, double degrees) =>
        new(t._celsius + degrees);

    public static Temperature operator -(Temperature t, double degrees) =>
        new(t._celsius - degrees);

    public static double operator -(Temperature a, Temperature b) =>
        a._celsius - b._celsius;

    // Comparison
    public static bool operator ==(Temperature a, Temperature b) =>
        Math.Abs(a._celsius - b._celsius) < 0.001;
    public static bool operator !=(Temperature a, Temperature b) => !(a == b);
    public static bool operator <(Temperature a, Temperature b) => a._celsius < b._celsius;
    public static bool operator >(Temperature a, Temperature b) => a._celsius > b._celsius;
    public static bool operator <=(Temperature a, Temperature b) => a._celsius <= b._celsius;
    public static bool operator >=(Temperature a, Temperature b) => a._celsius >= b._celsius;

    // Implicit conversions
    public static implicit operator double(Temperature t) => t._celsius;

    public override bool Equals(object? obj) =>

    public override int GetHashCode() => _celsius.GetHashCode();
```

### ตัวอย่างที่ 3: IAdditionOperators และ Generic Operators (C# 11+)

```csharp
using System;
using System.Numerics;

// ใช้ static interface methods สำหรับ operator interfaces
public struct Percentage : 
    IAdditionOperators<Percentage, Percentage, Percentage>,
    ISubtractionOperators<Percentage, Percentage, Percentage>,
    IMultiplyOperators<Percentage, double, Percentage>,
    IComparisonOperators<Percentage, Percentage, bool>
{
    public double Value { get; }

    public Percentage(double value) => Value = value;

    // static interface implementations
    public static Percentage operator +(Percentage left, Percentage right) =>
        new(left.Value + right.Value);

    public static Percentage operator -(Percentage left, Percentage right) =>
        new(left.Value - right.Value);

    public static Percentage operator *(Percentage left, double right) =>
        new(left.Value * right);

    public static bool operator ==(Percentage left, Percentage right) =>
        Math.Abs(left.Value - right.Value) < 0.0001;

    public static bool operator !=(Percentage left, Percentage right) =>
        !(left == right);

    public static bool operator <(Percentage left, Percentage right) =>
        left.Value < right.Value;

    public static bool operator >(Percentage left, Percentage right) =>
        left.Value > right.Value;

    public static bool operator <=(Percentage left, Percentage right) =>
        left.Value <= right.Value;

    public static bool operator >=(Percentage left, Percentage right) =>
        left.Value >= right.Value;

    // Implicit conversion
    public static implicit operator Percentage(double value) => new(value);
    public static explicit operator double(Percentage p) => p.Value;

    // Apply percentage to a value
    public decimal ApplyTo(decimal amount) => amount * (decimal)(Value / 100);

    public override string ToString() => $"{Value:F1}%";

    public override bool Equals(object? obj) =>
        obj is Percentage other && this == other;

    public override int GetHashCode() => Value.GetHashCode();
}

// Generic function ที่ใช้ IAdditionOperators
public static class Aggregator
{
    public static T Sum<T>(T[] values) where T : IAdditionOperators<T, T, T>
    {
        if (values.Length == 0)
            throw new ArgumentException("Array cannot be empty");
        
        T result = values[0];
        for (int i = 1; i < values.Length; i++)
            result += values[i];
        
        return result;
    }
}

class PercentageDemo
{
    static void Main()
    {
        Console.WriteLine("=== Percentage Type ===");
        
        Percentage tax = new(7);           // 7% VAT
        Percentage discount = new(10);     // 10% discount
        Percentage fee = new(2.5);         // 2.5% fee

        Console.WriteLine($"Tax: {tax}");
        Console.WriteLine($"Discount: {discount}");
        Console.WriteLine($"Tax + Fee: {tax + fee}");
        Console.WriteLine($"Discount - Tax: {discount - tax}");

        decimal price = 1000m;
        Console.WriteLine($"\nราคา: {price:C}");
        Console.WriteLine($"VAT ({tax}): {tax.ApplyTo(price):C}");
        Console.WriteLine($"ราคารวม VAT: {price + tax.ApplyTo(price):C}");
        Console.WriteLine($"ส่วนลด ({discount}): {discount.ApplyTo(price):C}");

        Console.WriteLine($"\ntax > fee: {tax > fee}");
        Console.WriteLine($"discount >= new Percentage(10): {discount >= new Percentage(10)}");

        Console.WriteLine("\n=== Generic Aggregator ===");
        
        var percentages = new Percentage[] { new(5), new(3), new(7), new(2) };
        var totalPercentage = Aggregator.Sum(percentages);
        Console.WriteLine($"Sum of percentages: {totalPercentage}");

        var intNumbers = new[] { 10, 20, 30, 40, 50 };
        var totalInt = Aggregator.Sum(intNumbers);
        Console.WriteLine($"Sum of ints: {totalInt}");
    }
}
```

### แนวทางปฏิบัติที่ดี

1. **Source Generators**: ใช้ Incremental Generator แทน ISourceGenerator สำหรับโปรเจ็กต์ใหม่เพราะมีประสิทธิภาพสูงกว่า
2. **Expression Trees**: เหมาะสำหรับ dynamic query building แต่ระวัง performance overhead
3. **Dynamic**: ใช้เฉพาะเมื่อจำเป็น เนื่องจากสูญเสีย compile-time type safety
4. **Unsafe Code**: ใช้เฉพาะเมื่อ performance critical และมีความเข้าใจ memory management
5. **Reflection**: Cache TypeInfo/MethodInfo เพื่อลด overhead หรือใช้ Source Generator แทน
6. **Generics**: ใช้ covariance/contravariance อย่างระมัดระวัง ให้ถูกต้องตาม Liskov Substitution Principle
7. **Pattern Matching**: ทำให้โค้ดอ่านง่ายขึ้นมาก ควรใช้แทน if-else chain
8. **Iterators**: ใช้ yield return สำหรับ lazy evaluation ประหยัด memory สำหรับข้อมูลขนาดใหญ่
9. **Operator Overloading**: ควรทำให้ semantics สอดคล้องกับ operators ทั่วไป และ implement IEquatable, IComparable
