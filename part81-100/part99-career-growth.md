## ขั้นตอนที่ 982: เตรียมสัมภาษณ์ .NET

```csharp
// คำถามที่พบบ่อยในการสัมภาษณ์ .NET

// 1. Value types vs Reference types
struct PointStruct { public int X, Y; }  // stack-allocated
class PointClass { public int X, Y; }    // heap-allocated

// struct: เหมาะกับข้อมูลเล็ก ๆ ที่ไม่ต้องการ inheritance
// class: เหมาะกับ complex objects, polymorphism

// 2. async/await internals
// State machine generated by compiler
// Task != Thread: Task is a unit of work, Thread is OS resource
// async void: ห้ามใช้ยกเว้น event handlers (ไม่สามารถ await ได้)
public async Task GoodAsync() { await Task.Delay(100); }
// public async void BadAsync() { await Task.Delay(100); } ← อันตราย

// 3. IDisposable pattern
public class ResourceManager : IDisposable
{
    private bool _disposed = false;
    private readonly UnmanagedResource _resource = new();
    
    protected virtual void Dispose(bool disposing)
    {
        if (!_disposed)
        {
            if (disposing)
                _resource?.Dispose(); // managed resources
            
            // free unmanaged resources here
            _disposed = true;
        }
    }
    
    public void Dispose()
    {
        Dispose(disposing: true);
        GC.SuppressFinalize(this); // ไม่ต้อง finalize อีก
    }
    
    ~ResourceManager() => Dispose(disposing: false); // safety net
}

// 4. Dependency Injection scopes
// Singleton: 1 instance ตลอด app lifetime
// Scoped: 1 instance per request (เหมาะสำหรับ DbContext)
// Transient: new instance ทุกครั้งที่ request

// CAPTIVE DEPENDENCY: Singleton inject Scoped = bug!
// ✅ แก้: inject IServiceScopeFactory แทน
public class BackgroundJob(IServiceScopeFactory scopeFactory)
{
    public async Task RunAsync()
    {
        using var scope = scopeFactory.CreateScope();
        var dbContext = scope.ServiceProvider.GetRequiredService<AppDbContext>();
        // use dbContext...
    }
}

// 5. SOLID ตัวอย่างสั้น ๆ
// Single Responsibility: class ทำหน้าที่เดียว
// Open/Closed: extend via interface, not modify
// Liskov: subclass ต้องแทนที่ parent ได้
// Interface Segregation: interface เล็ก ๆ specific
// Dependency Inversion: depend on abstraction (interface)
```

# How to contribute to .NET ecosystem

# 1. Find a good first issue
# github.com/dotnet/aspnetcore/issues?q=label:"good+first+issue"
# github.com/dotnet/runtime/issues?q=label:"help+wanted"

# 2. Fork, clone, setup

cd aspnetcore
git remote add upstream https://github.com/dotnet/aspnetcore.git

# 3. Create feature branch
git checkout -b fix/issue-12345-description

# 4. Make changes, follow contribution guide
# Read CONTRIBUTING.md first!
# Run tests: dotnet test

## ขั้นตอนที่ 990: Continuous Learning Strategy

```
30-60-90 Day Learning Plan:

Day 1-30 (Foundation):
□ Pick one area to deepen (e.g., performance optimization)
□ Read 1 technical book (Clean Architecture, Domain-Driven Design)
□ Build a small project applying concepts
□ Write 2 blog posts about what you learned

Day 31-60 (Practice):
□ Contribute to one open source project
□ Start a side project with proper architecture
□ Attend 1 meetup or conference
□ Complete 1 online course or certification path

Day 61-90 (Share):
□ Present at a meetup or internal tech talk
□ Write a comprehensive tutorial series
□ Mentor someone newer
□ Refactor old code with new knowledge

Yearly Goals:
- 12 blog posts (1/month)
- 2 certifications
- 1 open source contribution
- 1 conference talk (even local)
- 2 books read
- 1 side project launched
```
